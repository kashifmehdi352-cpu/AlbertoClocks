/* ================= VISITOR COUNTER ================= */

let visitors =
localStorage.getItem("albertoVisitors");

if(!visitors){

    visitors = 1;

}else{

    visitors = Number(visitors) + 1;

}

localStorage.setItem(
    "albertoVisitors",
    visitors
);

document.getElementById(
    "visitorCount"
).textContent = visitors;


/* ================= CART ================= */

let cart =
JSON.parse(
    localStorage.getItem("albertoCart")
) || [];


function updateCart(){

    document.getElementById(
        "cartCount"
    ).textContent = cart.length;

    let html = "";
    let total = 0;

    if(cart.length === 0){

        html = "Your cart is empty.";

    }else{

        cart.forEach(function(item,index){

            total += item.price;

            html += `
                <div style="
                    display:flex;
                    justify-content:space-between;
                    align-items:center;
                    border-bottom:1px solid #333;
                    padding:12px 0;
                ">

                    <span>
                        ${item.name}
                    </span>

                    <strong>
                        $${item.price}
                    </strong>

                    <button
                        onclick="removeCart(${index})"
                        style="
                            background:none;
                            border:1px solid #d4af37;
                            color:#d4af37;
                            padding:4px 8px;
                        "
                    >
                        ×
                    </button>

                </div>
            `;

        });

    }

    document.getElementById(
        "cartItems"
    ).innerHTML = html;

    document.getElementById(
        "cartTotal"
    ).textContent = total;

    localStorage.setItem(
        "albertoCart",
        JSON.stringify(cart)
    );

}


function addToCart(name,price){

    cart.push({
        name:name,
        price:price
    });

    updateCart();

    alert(
        name +
        " has been added to your cart."
    );

}


function removeCart(index){

    cart.splice(index,1);

    updateCart();

}


function openCart(){

    updateCart();

    const modal =
    new bootstrap.Modal(
        document.getElementById("cartModal")
    );

    modal.show();

}


function checkout(){

    if(cart.length === 0){

        alert("Your cart is empty.");

        return;

    }

    alert(
        "Thank you for shopping with Alberto Watches!"
    );

}

function filterCategory(category){

    alert(
        "Opening " +
        category +
        " collection..."
    );

    window.location.href =
    "products.html?category=" +
    encodeURIComponent(category);

}


/* ================= DATE TIME ================= */

function updateTicker(){

    const now = new Date();

    const date =
    now.toLocaleDateString();

    const time =
    now.toLocaleTimeString();

    document.getElementById(
        "tickerText"
    ).textContent =

    "ALBERTO WATCHES • " +
    "TIMELESS PRECISION • " +
    "DATE: " +
    date +
    " • TIME: " +
    time +
    " • PREMIUM WATCHES • ";

}

updateTicker();

setInterval(
    updateTicker,
    1000
);
